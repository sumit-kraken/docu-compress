class OrderProcessingService:
    def __init__(self, db_client, payment_gateway, notification_service):
        self.db = db_client
        self.payment = payment_gateway
        self.notifier = notification_service

    def process_customer_order(self, order_id: str, user_id: str) -> Dict[str, Any]:
        """
        Executes complete checkout transaction including inventory check,
        payment capture, DB record update, and receipt email generation.
        """
        # Validate order inventory
        order = self.db.get_order(order_id)
        if not order:
            raise ValueError("Order not found")
            
        for item in order.items:
            stock = self.db.check_stock(item.sku)
            if stock < item.quantity:
                raise OutOfStockException(f"Item {item.sku} unavailable")
                
        # Charge payment card
        payment_res = self.payment.charge(
            amount=order.total_price,
            currency="USD",
            customer_id=user_id
        )
        
        if payment_res.status != "SUCCESS":
            self.notifier.send_payment_failure(user_id, order_id)
            return {"status": "FAILED", "reason": payment_res.error_message}
            
        # Update order status
        self.db.update_status(order_id, "PAID")
        self.notifier.send_receipt(user_id, order_id)
        return {"status": "SUCCESS", "txn_id": payment_res.txn_id}
